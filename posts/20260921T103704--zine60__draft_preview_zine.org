  #+begin_details
  #+html: <summary>配置</summary>

  #+begin_src emacs-lisp
    (use-package repeat
      :hook (after-init . repeat-mode)
      :config
      (defvar-keymap spike-leung/char-repeat-map
        :doc "Repeat map for char related commands."
        :repeat t
        "b" #'backward-char
        "f" #'forward-char)
      (defvar-keymap spike-leung/word-repeat-map
        :doc "Repeat map for word related commands."
        :repeat t
        "b" #'backward-word
        "f" #'forward-word)
      (defvar-keymap spike-leung/paragraph-repeat-map
        :doc "Repeat map for paragraph related commands."
        :repeat t
        "{" #'backward-paragraph

#+begin_quote
俺也一樣是一個網絡迷因，出自央視版《三國演義》第一集桃園結義的劇情。
關羽三次用得體的語言向劉備表達自己心意，而張飛三次附和道「俺也一樣」。

「俺也一樣」是不是表達意見？是。
那麼「俺也一樣」對討論有沒有幫助？沒有。
因為它不提供任何新信息，而討論想要獲得進展，需要的是事實和論述。
所以不要在討論裡說「俺也一樣」。
#+end_quote
